    static Brush Lin(string a, string b, double angle = 90) { var g = new LinearGradientBrush(C(a), C(b), angle); g.Freeze(); return g; }
    static Brush Rad(string a, string b, double cx = 0.35, double cy = 0.3) { var g = new RadialGradientBrush(C(a), C(b)) { GradientOrigin = new Point(cx, cy), Center = new Point(0.5, 0.5), RadiusX = 0.6, RadiusY = 0.6 }; g.Freeze(); return g; }
    static Brush S(string hex) { var b = new SolidColorBrush(C(hex)); b.Freeze(); return b; }
    static Pen Pn(string hex, double w) { var p = new Pen(S(hex), w) { LineJoin = PenLineJoin.Round, StartLineCap = PenLineCap.Round, EndLineCap = PenLineCap.Round }; p.Freeze(); return p; }

    static void Gear(DrawingContext dc, string light, string dark)
    {
        var teeth = new StreamGeometry();
        using (var ctx = teeth.Open())
        {
            for (int i = 0; i < 16; i++)
            {
                double a = i * Math.PI / 8, a2 = (i + 1) * Math.PI / 8, r = i % 2 == 0 ? 14 : 10.5;
                var p1 = new Point(16 + r * Math.Cos(a), 16 + r * Math.Sin(a));
                var p2 = new Point(16 + r * Math.Cos(a2), 16 + r * Math.Sin(a2));
                if (i == 0) ctx.BeginFigure(p1, true, true); else ctx.LineTo(p1, true, true);
                ctx.LineTo(p2, true, true);
            }
        }
        teeth.Freeze();
        dc.DrawGeometry(Lin(light, dark), Pn("#101418", 1.2), teeth);
        dc.DrawEllipse(S("#101418"), null, new Point(16, 16), 4, 4);
    }

    static void Pixels(DrawingContext dc, string[] rows, string fill, double w = 32, double h = 32)
    {
        double cell = Math.Min(w / rows[0].Length, h / rows.Length);
        double ox = (w - cell * rows[0].Length) / 2, oy = (h - cell * rows.Length) / 2;
        for (int y = 0; y < rows.Length; y++)
            for (int x = 0; x < rows[y].Length; x++)
                if (rows[y][x] == 'X') dc.DrawRectangle(S(fill), null, new Rect(ox + x * cell, oy + y * cell, cell, cell));
    }

    static BitmapSource Render(int width, int height, double canvasW, double canvasH, Action<DrawingContext> draw, bool pixel)
    {
        var dv = new DrawingVisual();
        if (pixel) RenderOptions.SetEdgeMode(dv, EdgeMode.Aliased);
        using (var dc = dv.RenderOpen())
        {
            dc.PushTransform(new ScaleTransform(width / canvasW, height / canvasH));
            draw(dc);
            dc.Pop();
        }
        var bmp = new RenderTargetBitmap(width, height, 96, 96, PixelFormats.Pbgra32);
        bmp.Render(dv);
        bmp.Freeze();
        return bmp;
    }

    static void Save(BitmapSource b, string path)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        var e = new PngBitmapEncoder();
        e.Frames.Add(BitmapFrame.Create(b));
        using var f = File.Create(path);
        e.Save(f);
    }

    [STAThread]
    static void Main(string[] a)
    {
        string root = a[0];
        int[] sizes = { 16, 24, 32, 48 };
        var sheet = new List<(string label, BitmapSource img)>();
        foreach (var (group, set) in new[] { ("start", Start), ("quicklaunch", Quick) })
            foreach (var icon in set)
                foreach (int s in sizes)
                {
                    var img = Render(s, s, 32, 32, icon.Value, icon.Key.StartsWith("pixel"));
                    Save(img, Path.Combine(root, group, s + "px", icon.Key + ".png"));
                    sheet.Add(($"{group}/{icon.Key}", img));
                }
        foreach (var t in Toggle)
            foreach (int scale in new[] { 1, 2 })
            {
                int mult = t.Key.StartsWith("pixel") ? scale * 2 : scale; // the pixel arrow is drawn on a 6x8 grid
                var img = Render(t.Value.w * mult, t.Value.h * mult, t.Value.w, t.Value.h, t.Value.draw, t.Key.StartsWith("pixel"));
                Save(img, Path.Combine(root, "toggle", scale == 1 ? "100%" : "200%", t.Key + ".png"));
                sheet.Add(($"toggle/{t.Key}", img));
            }

        // Contact sheet: every icon at every size, on light and dark backgrounds.
        var names = sheet.Select(x => x.label).Distinct().ToList();
        int rowH = 56, width = 560;
        var dv = new DrawingVisual();
        using (var dc = dv.RenderOpen())
        {
            dc.DrawRectangle(Brushes.White, null, new Rect(0, 0, width, names.Count * rowH));
            for (int i = 0; i < names.Count; i++)
            {
                dc.DrawRectangle(S(i % 2 == 0 ? "#2b2f33" : "#3a3f44"), null, new Rect(width / 2 + 40, i * rowH, width / 2 - 40, rowH));
                dc.DrawText(new FormattedText(names[i], System.Globalization.CultureInfo.InvariantCulture, FlowDirection.LeftToRight, new Typeface("Segoe UI"), 11, Brushes.Black, 1), new Point(4, i * rowH + 20));
                double x = 120;
                foreach (var img in sheet.Where(s => s.label == names[i]).Select(s => s.img))
                {
                    dc.DrawImage(img, new Rect(x, i * rowH + 4, img.PixelWidth, img.PixelHeight));
                    dc.DrawImage(img, new Rect(x + width / 2 - 80, i * rowH + 4, img.PixelWidth, img.PixelHeight));
                    x += Math.Max(img.PixelWidth + 4, 20);
                }
            }
        }
        var bmp = new RenderTargetBitmap(width, names.Count * rowH, 96, 96, PixelFormats.Pbgra32);
        bmp.Render(dv);
        Save(bmp, Path.Combine(a[1], "icon-sheet.png"));
        Console.WriteLine($"{sheet.Count} images, {names.Count} icons");
    }
